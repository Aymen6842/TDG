% --- Auto-resolving figure macro -------------------------------------------
% Path is passed WITHOUT extension; tries .pdf, .png, .jpg, else a loud pending box.
\newcommand{\reportfigure}[3]{%
  \begin{figure}[H]\centering
  \IfFileExists{#1.pdf}{\includegraphics[width=0.85\textwidth]{#1.pdf}}{%
  \IfFileExists{#1.png}{\includegraphics[width=0.85\textwidth]{#1.png}}{%
  \IfFileExists{#1.jpg}{\includegraphics[width=0.85\textwidth]{#1.jpg}}{%
  \fbox{\begin{minipage}{0.8\textwidth}\centering\vspace{10pt}{\color{RoyalPurple}\bfseries\Large FIGURE PENDING}\\[6pt]Expected file: \texttt{#1.pdf}, \texttt{.png} or \texttt{.jpg}\\[4pt]{\small #2}\vspace{10pt}\end{minipage}}}}}
  \caption{#2}\label{fig:#3}\end{figure}}
