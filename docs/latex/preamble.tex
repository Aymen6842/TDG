  \noindent{\fontsize{17}{21}\selectfont\bfseries\color{chapterblue}#1}\\[0.45em]%
  \noindent{\color{headerrule}\rule{\textwidth}{0.8pt}}%
}

% --- Auto-resolving figure macro -------------------------------------------
% \reportfigure{path}{caption}{label}           — default 0.85\textwidth
% \reportfigure[0.5\textwidth]{path}{caption}{label} — custom width
% Both width AND height are capped (keepaspectratio), so tall diagrams no
% longer explode and wide ones no longer dictate their own scale.
\newcommand{\figmaxheight}{0.42\textheight}
\newcommand{\reportfigure}[4][0.85\textwidth]{%
  \begin{figure}[H]\centering
  \IfFileExists{#2.pdf}{\includegraphics[width=#1,height=\figmaxheight,keepaspectratio]{#2.pdf}}{%
  \IfFileExists{#2.png}{\includegraphics[width=#1,height=\figmaxheight,keepaspectratio]{#2.png}}{%
  \IfFileExists{#2.jpg}{\includegraphics[width=#1,height=\figmaxheight,keepaspectratio]{#2.jpg}}{%
  \fbox{\begin{minipage}{0.8\textwidth}\centering\vspace{10pt}{\color{RoyalPurple}\bfseries\Large FIGURE PENDING}\\[6pt]{\small #3}\vspace{10pt}\end{minipage}}}}}
  \caption{#3}\label{fig:#4}\end{figure}}

% --- Full-page sideways figure macro -----------------------------------------
% For diagrams too wide to read in portrait: a dedicated landscape page.
% \reportsideways{path}{caption}{label}
\newcommand{\reportsideways}[3]{%
  \begin{sidewaysfigure}\centering
  \IfFileExists{#1.pdf}{\includegraphics[width=0.96\textheight,height=0.92\textwidth,keepaspectratio]{#1.pdf}}{%
  \IfFileExists{#1.png}{\includegraphics[width=0.96\textheight,height=0.92\textwidth,keepaspectratio]{#1.png}}{%
  \fbox{\begin{minipage}{0.8\textwidth}\centering\vspace{10pt}{\color{RoyalPurple}\bfseries\Large FIGURE PENDING}\\[6pt]{\small #2}\vspace{10pt}\end{minipage}}}}
  \caption{#2}\label{fig:#3}\end{sidewaysfigure}}

% --- Screenshot macro -------------------------------------------------------
% Same auto-resolve behaviour as \reportfigure, plus a thin gray frame so
% white UI captures do not bleed into the page.
\newcommand{\reportscreenshot}[4][0.9\textwidth]{%
  \begin{figure}[H]\centering
  {\setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.4pt}\color{screenshotframe}%
  \IfFileExists{#2.png}{\fbox{\includegraphics[width=#1,height=\figmaxheight,keepaspectratio]{#2.png}}}{%
  \IfFileExists{#2.jpg}{\fbox{\includegraphics[width=#1,height=\figmaxheight,keepaspectratio]{#2.jpg}}}{%
  \color{black}\fbox{\begin{minipage}{0.8\textwidth}\centering\vspace{10pt}{\color{RoyalPurple}\bfseries\Large SCREENSHOT PENDING}\\[6pt]{\small #3}\vspace{10pt}\end{minipage}}}}}
  \caption{#3}\label{fig:#4}\end{figure}}
